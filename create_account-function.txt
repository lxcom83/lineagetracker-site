function cors(req) {
  const o = req.headers.get('origin') || '';
  return {
    'Access-Control-Allow-Origin': ORIGINS.includes(o) ? o : ORIGINS[0],
    'Access-Control-Allow-Headers': 'content-type, authorization, apikey, x-client-info',
    'Access-Control-Allow-Methods': 'POST, OPTIONS',
    'Vary': 'Origin',
  };
}
function reply(req, status, obj) {
  return new Response(JSON.stringify(obj), { status, headers: { ...cors(req), 'Content-Type': 'application/json', 'Cache-Control': 'no-store' } });
}
// Reads a small JSON body.  Returns the object, or a Response to send back.
async function readJson(req, max = 20000) {
  if (req.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors(req) });
  if (req.method !== 'POST') return reply(req, 405, { ok: false, error: 'method' });
  try {
    if (Number(req.headers.get('content-length') || 0) > max * 4) return reply(req, 413, { ok: false, error: 'too_big' });
    const t = await req.text();
    if (t.length > max) return reply(req, 413, { ok: false, error: 'too_big' });
    const d = JSON.parse(t);
    if (!d || typeof d !== 'object' || Array.isArray(d)) return reply(req, 400, { ok: false, error: 'bad_request' });
    return d;
  } catch { return reply(req, 400, { ok: false, error: 'bad_request' }); }
}

}

// Who is signed in?  Asks Supabase Auth to check the token, so a forged or expired token is refused.
async function userFromJwt(req) {
  const h = req.headers.get('authorization') || '';
  const m = /^Bearer ([A-Za-z0-9._-]{20,4000})$/.exec(h);
  if (!m) return null;
  const r = await fetch(`${SB}/auth/v1/user`, { headers: { apikey: Deno.env.get('SUPABASE_ANON_KEY') || KEY, Authorization: 'Bearer ' + m[1] } });
  if (!r.ok) return null;
  const u = await r.json();
  if (!u || !u.id) return null;
  let claims = {};
  try { claims = JSON.parse(new TextDecoder().decode(Uint8Array.from(atob(m[1].split('.')[1].replace(/-/g, '+').replace(/_/g, '/')), (c) => c.charCodeAt(0)))); } catch { /* the token was already accepted above */ }
  return { id: u.id, email: u.email, aal: claims.aal || 'aal1', session: claims.session_id || null };
}

function b64url(bytes) {
  let s = ''; for (const b of bytes) s += String.fromCharCode(b);
  return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}
async function sha256hex(text) {
  const d = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(text));
  return [...new Uint8Array(d)].map((b) => b.toString(16).padStart(2, '0')).join('');
}

async function hmac(keyBytes, text) {
  const k = await crypto.subtle.importKey('raw', keyBytes, { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  return new Uint8Array(await crypto.subtle.sign('HMAC', k, new TextEncoder().encode(text)));
}

// IPv6 people can switch between billions of addresses in their own block, so limits count the whole /64 block.
function ipBucket(ip) {
  if (!ip.includes(':')) return ip;
  const [head, tail = ''] = ip.split('::'); const h = head ? head.split(':') : [], t = tail ? tail.split(':') : [];
  const full = [...h, ...Array(Math.max(0, 8 - h.length - t.length)).fill('0'), ...t];
  return full.slice(0, 4).map((x) => x.toLowerCase().replace(/^0+(?=.)/, '')).join(':') + '::/64';
}

// Counts a hit against a limit.  Returns true while allowed.  Keys are keyed hashes (HMAC with a server secret), so what is
// stored cannot be turned back into an address or email by guessing.  Fails closed.
async function allow(kind, who, limit, windowSeconds) {
  try { return (await rpc('acct_rl_hit', { p_key: b64url(await hmac(await limitKey(), kind + ':' + who)), p_limit: limit, p_window_seconds: windowSeconds })) === true; }
  catch { return false; }
}

// May we send this address another account email?  Counts per (address, visitor) and per address overall, so one visitor
// cannot use us to flood someone.  A refusal is visible to the visitor and tells them nothing about the address.
async function mailGate(kind, email, ip, perVisitor = 3) {
  const k = emailKey(email);
  if (!(await allow(kind + '-pair', k + '|' + ipBucket(ip), perVisitor, 3600))) return false;
  return allow('mail-email', k, 8, 3600);
}
// A daily ceiling on mails really sent, per kind, to protect the sending domain.  Only mails that are about to go out count
// (junk requests for addresses with no account use none of it), and it is asked inside the background job, where a refusal
// cannot be seen by whoever asked.  Hitting it is written to the audit log (once an hour) so staff can see it.
async function mailBudget(kind) {
  const cap = Number(Deno.env.get('MAIL_DAILY_CAP')) || ({ signup: 200, reset: 100, resend: 100, blog: 100 }[kind] || 100);
  if (await allow('mail-day-' + kind, 'all', cap, 86400)) return true;
  if (await allow('mail-cap-note', kind, 1, 3600)) await rpc('acct_audit', { p_actor: null, p_target: null, p_action: 'mail_cap_reached', p_detail: { kind } }).catch(() => {});
  console.error('daily mail cap reached', kind); return false;
}
// Send an account mail if today's budget allows.  Returns true if it was sent.  Throws if the mail service refuses it.
async function sendAccountMail(kind, to, mail) {
  if (!(await mailBudget(kind))) return false;

}

async function turnstileOk(token, ip) {
  const secret = Deno.env.get('TURNSTILE_SECRET_KEY');
  if (!secret || !token || typeof token !== 'string' || token.length > 4096) return false;
  const body = new URLSearchParams({ secret, response: token });
  if (ip) body.set('remoteip', ip);
  try {
    const r = await fetch('https://challenges.cloudflare.com/turnstile/v0/siteverify', { method: 'POST', body });
    if (!r.ok) return false;
    const j = await r.json();
    // The check must have been passed on one of our own pages, not on someone else's page that uses the same site key.
    return j.success === true && typeof j.hostname === 'string' && ORIGINS.some((o) => { try { return new URL(o).hostname === j.hostname; } catch { return false; } });
  } catch { return false; }
}

function passwordProblem(pw, ctx) {
  if (typeof pw !== 'string') return 'password_short';
  if ([...pw].length < 10) return 'password_short';
  if (new TextEncoder().encode(pw).length > 72) return 'password_long';
  const low = pw.toLowerCase();
  if (COMMON.has(low)) return 'password_common';
  if (new Set([...low]).size <= 3) return 'password_simple';
  if (/^(0123456789|1234567890|abcdefghij|qwertyuiop|asdfghjkl)/.test(low)) return 'password_simple';
  const local = String((ctx && ctx.email) || '').split('@')[0].toLowerCase();
  if ((local.length >= 4 && low.includes(local)) || (ctx && ctx.alias && ctx.alias.length >= 4 && low.includes(String(ctx.alias).toLowerCase()))) return 'password_personal';
  return null;
}
// Has this password appeared in a known data breach?  Only the first 5 characters of its SHA-1 are sent.  If the service is down, we let it through.
async function passwordPwned(pw) {
  try {
    const d = await crypto.subtle.digest('SHA-1', new TextEncoder().encode(pw));
    const hex = [...new Uint8Array(d)].map((b) => b.toString(16).padStart(2, '0')).join('').toUpperCase();
    const r = await fetch('https://api.pwnedpasswords.com/range/' + hex.slice(0, 5), { headers: { 'Add-Padding': 'true' }, signal: AbortSignal.timeout(3000) });
    if (!r.ok) return false;
    const rest = hex.slice(5);
    return (await r.text()).split('\n').some((l) => { const [h, n] = l.trim().split(':'); return h === rest && Number(n) > 0; });
  } catch { return false; }
}
async function passwordError(pw, ctx) { return passwordProblem(pw, ctx) || ((await passwordPwned(pw)) ? 'password_pwned' : null); }

async function aliasSig(alias, exp) { return b64url(await hmac(await aliasKey(), alias + '|' + exp)); }
async function aliasVerify(a) {
  if (!a || typeof a.alias !== 'string' || typeof a.sig !== 'string' || !Number.isInteger(a.exp)) return false;
  if (!ALIAS_RE.test(a.alias) || a.exp < Math.floor(Date.now() / 1000)) return false;
  const want = await aliasSig(a.alias, a.exp);
  if (want.length !== a.sig.length) return false;
  let diff = 0; for (let i = 0; i < want.length; i++) diff |= want.charCodeAt(i) ^ a.sig.charCodeAt(i);
  return diff === 0;
}

const stHeaders = (type) => ({ apikey: KEY, Authorization: 'Bearer ' + KEY, ...(type ? { 'Content-Type': type } : {}) });
async function storagePut(bucket, path, bytes, upsert = false) {
  const r = await fetch(`${SB}/storage/v1/object/${bucket}/${path}`, { method: 'POST', headers: { ...stHeaders('image/jpeg'), 'x-upsert': upsert ? 'true' : 'false', 'cache-control': 'max-age=31536000' }, body: bytes });
  if (!r.ok) throw new Error('storage put ' + r.status);
}
async function storageGet(bucket, path) {
  const r = await fetch(`${SB}/storage/v1/object/${bucket}/${path}`, { headers: stHeaders() });
  if (!r.ok) throw new Error('storage get ' + r.status);
  return new Uint8Array(await r.arrayBuffer());
}
async function storageDelete(bucket, paths) {
  if (!paths.length) return;
  const r = await fetch(`${SB}/storage/v1/object/${bucket}`, { method: 'DELETE', headers: stHeaders('application/json'), body: JSON.stringify({ prefixes: paths }) });
  if (!r.ok) console.error('storage delete', bucket, r.status);
}
async function storageSign(path) {
  const r = await fetch(`${SB}/storage/v1/object/sign/${PRIVATE}/${path}`, { method: 'POST', headers: stHeaders('application/json'), body: JSON.stringify({ expiresIn: 3600 }) });
  if (!r.ok) return null;
  const j = await r.json();
  return j && j.signedURL ? `${SB}/storage/v1${j.signedURL}` : null;
}
const publicPath = (p) => p.replace(/^[0-9a-f-]{36}\//, '');
async function withUrls(images) { return Promise.all((images || []).map(async (i) => ({ ...i, url: await storageSign(i.path) }))); }
// Blog pictures of a member, asked for before their account is deleted (empty if the blog tables are not installed yet).
async function blogPathsOf(member) { try { return (await rpc('acct_blog_paths_of', { p_member: member })) || []; } catch { return []; } }
async function blogCleanup(paths) {
  await storageDelete(PRIVATE, paths.map((x) => x.path));
  await storageDelete(PUBLIC, paths.filter((x) => x.public).map((x) => publicPath(x.path)));
  try { await rpc('acct_blog_recount', {}); } catch { /* blog not installed */ }
}

// A normal sign-up chooses NO password here: the login gets a random one nobody knows, and the person chooses their real password
// when they open the link we email them.  So nobody can register someone else's address and know the password.
// A person arriving from an invitation link has already proved they own the address, so they choose a password on the sign-up page.
Deno.serve(async (req) => {
  const d = await readJson(req); if (isResponse(d)) return d;
  const ip = ipOf(req) || 'unknown';
  const email = cleanEmail(d.email);
  if (!emailOk(email)) return reply(req, 400, { ok: false, error: 'email' });
  if (d.over18 !== true) return reply(req, 400, { ok: false, error: 'over18' });
  if (typeof d.terms !== 'string' || d.terms.length > 60) return reply(req, 400, { ok: false, error: 'terms' });
  const country = d.country == null || d.country === '' ? null : String(d.country);
  const region = d.region == null || d.region === '' ? null : String(d.region);
  if (country && !COUNTRY_RE.test(country)) return reply(req, 400, { ok: false, error: 'country' });
  if (region && (!country || !REGION_RE.test(region))) return reply(req, 400, { ok: false, error: 'region' });
  if (!(await allow('signup-ip', ipBucket(ip), 10, 3600))) return reply(req, 429, { ok: false, error: 'slow_down' });

  try {
    const settings = await rpc('public_settings', {});
    if (!settings || settings.signup_open !== true) return reply(req, 403, { ok: false, error: 'closed' });
    if (d.terms !== settings.terms_current) return reply(req, 409, { ok: false, error: 'terms_changed' });
    if (!(await turnstileOk(d.turnstile, ip))) return reply(req, 400, { ok: false, error: 'turnstile' });
    if (!(await aliasVerify(d.alias))) return reply(req, 400, { ok: false, error: 'alias' });
    const alias = d.alias.alias;

    const [sOk, iOk] = await Promise.all([activeKeys('service_options'), activeKeys('interest_options')]);
    // Only a person arriving with an invitation (which proves the address is theirs) can set country, emails and interests while
    // signing up.  Everyone else sets them on their account page once the address is confirmed: nobody can sign up an address
    // they do not own for newsletters.
    const hasInvite = d.invite != null && d.invite !== '';
    const services = hasInvite ? pickKeys(d.services, sOk, 10) : [], interests = hasInvite ? pickKeys(d.interests, iOk, 100) : [];

    let inviteHash = null;
    if (d.invite != null && d.invite !== '') {
      if (typeof d.invite !== 'string' || !/^[A-Za-z0-9_-]{20,100}$/.test(d.invite)) return reply(req, 400, { ok: false, error: 'invite_invalid' });
      inviteHash = await sha256hex(d.invite);
      const invitedEmail = await rpc('acct_invite_check', { p_hash: inviteHash });
      if (!invitedEmail) return reply(req, 400, { ok: false, error: 'invite_invalid' });
      if (cleanEmail(invitedEmail) !== email) return reply(req, 400, { ok: false, error: 'invite_email' });
    }
    const password = npw(d.password);
    if (inviteHash) { const bad = await passwordError(password, { email, alias }); if (bad) return reply(req, 400, { ok: false, error: bad }); }

    if ((await rpc('acct_alias_taken', { p_aliases: [alias] })).length) return reply(req, 409, { ok: false, error: 'alias_taken' });
    // Only a sign-up that will cause an email counts against the email limits (a mistake on the form does not use up the tries).
    // An invited sign-up sends none, so it is limited by the visitor's address alone (above).
    if (!inviteHash && !(await mailGate('signup', email, ip))) return reply(req, 429, { ok: false, error: 'slow_down' });
    const state = await rpc('acct_email_state', { p_email: email });

    // What happens next is the same job for everyone; for a normal sign-up it runs after the answer has gone back, so the
    // answer is identical (and equally quick) whether or not the address already has an account.
    const job = async () => {
      // Someone already has an account with this address: tell the owner of the address, not the visitor.
      if (state.state === 'active') {
        let sent = false;
        if (await allow('exists-mail', emailKey(email), 1, 1800)) sent = await sendAccountMail('signup', email, MAIL.exists()).catch((e) => { console.error('mail', String(e)); return false; });
        console.log('signup job: exists, mail ' + (sent ? 'sent' : 'not sent')); return { state: 'check_email' };
      }
      // A sign-up is waiting for its owner to open the link: send a fresh link and keep everything else exactly as it was, so a
      // stranger typing the address cannot overwrite or kill what is waiting for the owner.
      if (state.state === 'pending' && !inviteHash) {
        const token = newToken(); let sent = false;
        if (await rpc('acct_confirm_reissue', { p_email: email, p_hash: await sha256hex(token) })) sent = await sendAccountMail('signup', email, MAIL.confirm(`${SITE}/account-confirm.html#t=${token}`)).catch((e) => { console.error('mail', String(e)); return false; });
        console.log('signup job: pending, new link ' + (sent ? 'sent' : 'not sent')); return { state: 'check_email' };
      }
      // A leftover login that was never confirmed or used (or the pending one of someone arriving with an invitation, which proves
      // the address is theirs) is replaced.  A login that was confirmed or used is never touched.
      if (state.state === 'pending' || state.state === 'orphan') {
        const del = await gotrue('DELETE', '/admin/users/' + state.id); if (!del.ok && del.status !== 404) glog('old login not removed', del);
      }
      const made = await gotrue('POST', '/admin/users', { email, password: inviteHash ? password : randomPassword(), email_confirm: !!inviteHash, user_metadata: {} });
      if (!made.ok) { glog('createUser failed', made); console.log('signup job: failed (login)'); return { error: made.status === 422 ? 'none' : 'server' }; }
      const id = made.body.id;
      const token = inviteHash ? null : newToken();
      try {
        await rpc('acct_member_create', {
          p_id: id, p_alias: alias, p_country: inviteHash ? country : null, p_region: inviteHash ? region : null, p_terms: d.terms, p_services: services, p_interests: interests,
          p_active: !!inviteHash, p_confirm_hash: token ? await sha256hex(token) : null, p_invite_hash: inviteHash,
        });
      } catch (e) {
        await gotrue('DELETE', '/admin/users/' + id);
        const m = dbMsg(e); console.log('signup job: failed (member: ' + (m || 'db') + ')');
        if (m === 'alias_taken' || (e instanceof DbError && e.body && e.body.code === '23505')) return { error: 'alias_taken' };
        if (m === 'terms_changed') return { error: 'terms_changed' };
        if (m === 'invite_invalid') return { error: 'invite_invalid' };
        console.error('member_create failed', m); return { error: 'server' };
      }
      if (!inviteHash) {
        try {
          if (!(await sendAccountMail('signup', email, MAIL.confirm(`${SITE}/account-confirm.html#t=${token}`)))) throw new Error('daily mail cap');
        } catch (e) { console.error('confirm mail failed', String(e)); await gotrue('DELETE', '/admin/users/' + id); console.log('signup job: failed (mail)'); return { error: 'mail' }; }
      }
      console.log('signup job: created' + (inviteHash ? ' (invited)' : ', confirmation mail sent'));
      return { state: inviteHash ? 'active' : 'check_email' };
    };

    if (!inviteHash) { bg(job()); return reply(req, 200, { ok: true, state: 'check_email' }); }
    // An invitation proves the person owns the address, so any problem can be told to them straight.
    const out = await job();
    if (out.error === 'alias_taken' || out.error === 'terms_changed') return reply(req, 409, { ok: false, error: out.error });
    if (out.error === 'invite_invalid') return reply(req, 400, { ok: false, error: 'invite_invalid' });
    if (out.error) return reply(req, 502, { ok: false, error: 'server' });
    return reply(req, 200, { ok: true, state: out.state });
  } catch (e) { console.error('create_account failed', String(e)); return reply(req, 502, { ok: false, error: 'server' }); }
});
